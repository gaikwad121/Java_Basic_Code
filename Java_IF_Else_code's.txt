package Java_Practice;
import java.util.Scanner;
public class Java_Practice {
public static void main(String[] args) {

	Scanner sc = new Scanner(System.in);
    System.out.println("Enter any number");
    int num = sc.nextInt();
    
    // Check for valid positive integer input
    if (num <= 0) {
        System.out.println("Please enter a valid positive number");
    } else {
        // Check if divisible by 5 or 3
        if (num % 5 == 0 || num % 3 == 0) {
            // Nested check: if not divisible by 15
            if (num % 15 != 0) {
                System.out.println("The number is divisible by 5 or 3 but not divisible by 15");
            } else {
                System.out.println("The number does not meet the criteria");
            }
        } else {
            System.out.println("The number does not meet the criteria");
        }
    }
}
}
